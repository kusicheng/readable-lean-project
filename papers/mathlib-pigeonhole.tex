\section{Finite sets}
\begin{lemma}[{A fiber with weight above the bound for maps into the target}]\label{theorem:1}
Let \(\alpha\), \(\beta\), and \(M\) be types. Assume equality on \(\beta\) is decidable. Let \(s\) be a finite subset of \(\alpha\). Let \(t\) be a finite subset of \(\beta\). Let \(f\) be a function of type \(\alpha \to \beta\). Let \(w\) be a function of type \(\alpha \to M\). Let \(b\) be an element of \(M\). Equip \(M\) with a commutative additive monoid structure and a linear order. Assume \(\text{translation preserves and reflects order on }M\). Assume \(\forall a\, :\, \alpha,\; a \in s \implies f\left(a\right) \in t\). Assume \(\left|t\right| \mathbin{\bullet} b < \sum_{x\in s}\left(w\left(x\right)\right)\). Then
\begin{center}
\(\displaystyle \exists y\, :\, \beta,\; y \in t \land b < \sum_{x\in \{x\in s \mid f(x) = y\}}(w(x)).\)
\end{center}
\end{lemma}

\begin{proof}
Use the hypothesis \(\text{translation preserves and reflects order on }M\).
Use the hypothesis \(\left|t\right| \mathbin{\bullet} b < \sum_{x\in s}\left(w\left(x\right)\right)\).
Rewriting with Prerequisite~P\ref{premise:46696e7365742e73756d5f636f6e7374} and Prerequisite~P\ref{premise:46696e7365742e73756d5f6669626572776973655f6f665f6d6170735f746f} (using \(\forall a\, :\, \alpha,\; a \in s \implies f\left(a\right) \in t\)) in the preceding result gives \(\sum_{i\in t}(b) < \sum_{i\in t}(\sum_{x\in \{x\in s \mid f(x) = i\}}(w(x)))\).
By Prerequisite~P\ref{premise:46696e7365742e6578697374735f6c745f6f665f73756d5f6c74}, we obtain \(\exists y\, :\, \beta,\; y \in t \land b < \sum_{x\in \{x\in s \mid f(x) = y\}}(w(x))\).

\end{proof}

By Prerequisite~P\ref{premise:46696e7365742e6578697374735f6c745f6f665f73756d5f6c74}, we obtain \(\exists y\, :\, \beta,\; y \in t \land b < \sum_{x\in \{x\in s \mid f(x) = y\}}(w(x))\).

\end{proof}

\begin{proof}

\begin{proofstep}

\end{proofstep}

\end{proof}
