\section{Results}
\begin{theorem}[{A fiber with weight above the bound for maps into the target}]\label{theorem:1}
Let \(s\) and \(t\) be finite subsets of \(\mathbb{N}\). Let \(f\) and \(w\) be functions of type \(\mathbb{N} \to \mathbb{N}\). Let \(b\) be a natural number. Assume \(\forall x\, :\, \mathbb{N},\; x \in s \implies f\left(x\right) \in t\). Assume \(\left|t\right| \cdot b < \sum_{x\in s}\left(w\left(x\right)\right)\). Then
\begin{center}
\(\displaystyle \exists y\, :\, \mathbb{N},\; y \in t \land b < \sum_{x\in \{x\in s \mid f(x) = y\}}(w(x)).\)
\end{center}
\end{theorem}

\begin{proof}
By \paperlink{premise:4e61742e696e737449734f72646572656443616e63656c4164644d6f6e6f6964}{Prerequisite~P\ref{premise:4e61742e696e737449734f72646572656443616e63656c4164644d6f6e6f6964}}, we obtain \(\text{translation preserves and reflects order on }\mathbb{N}\).
By \paperlink{premise:46696e7365742e73756d5f636f6e7374}{Prerequisite~P\ref{premise:46696e7365742e73756d5f636f6e7374}} with \(\iota = \mathbb{N}\) and \(M = \mathbb{N}\), we obtain \(\sum_{x\in t}\left(b\right) = \left|t\right| \cdot b\).
Use the hypothesis \(\left|t\right| \cdot b < \sum_{x\in s}\left(w\left(x\right)\right)\).
By transitivity of the displayed relations, we obtain \(\sum_{x\in t}\left(b\right) < \sum_{x\in s}\left(w\left(x\right)\right)\).
Use the hypothesis \(\forall x\, :\, \mathbb{N},\; x \in s \implies f\left(x\right) \in t\).
By \paperlink{premise:46696e7365742e73756d5f6669626572776973655f6f665f6d6170735f746f}{Prerequisite~P\ref{premise:46696e7365742e73756d5f6669626572776973655f6f665f6d6170735f746f}} with \(\iota = \mathbb{N}\) and \(\kappa = \mathbb{N}\) and \(M = \mathbb{N}\) and \(w\) in place of its parameter \(f\), we obtain \(\sum_{j\in t}(\sum_{i\in \{i\in s \mid f(i) = j\}}(w(i))) = \sum_{i\in s}(w(i))\).
Reversing the equality, we obtain \(\sum_{i\in s}(w(i)) = \sum_{j\in t}(\sum_{i\in \{i\in s \mid f(i) = j\}}(w(i)))\).
By transitivity of the displayed relations, we obtain \(\sum_{x\in t}(b) < \sum_{y\in t}(\sum_{x\in \{x\in s \mid f(x) = y\}}(w(x)))\).
By \paperlink{premise:46696e7365742e6578697374735f6c745f6f665f73756d5f6c74}{Prerequisite~P\ref{premise:46696e7365742e6578697374735f6c745f6f665f73756d5f6c74}} with \(\iota = \mathbb{N}\) and \(M = \mathbb{N}\), we obtain \(\exists y\, :\, \mathbb{N},\; y \in t \land b < \sum_{x\in \{x\in s \mid f(x) = y\}}(w(x))\).
\end{proof}
