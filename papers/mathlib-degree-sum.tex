Let \(j\) be a function of type \(\Pi_{a\, :\, \beta}\, a \in t \to \alpha\). Let \(\mathit{hi}\) be a proof of \(\forall a\, :\, \alpha,\; \forall \mathit{ha}\, :\, a \in s,\; i\left(a, \mathit{ha}\right) \in t\). Let \(\mathit{hj}\) be a proof of \(\forall a\, :\, \beta,\; \forall \mathit{ha}\, :\, a \in t,\; j\left(a, \mathit{ha}\right) \in s\). Assume \(\forall a\, :\, \alpha,\; \forall \mathit{ha}\, :\, a \in s,\; j(i(a,\allowbreak{} \mathit{ha}),\allowbreak{} \mathit{hi}(a,\allowbreak{} \mathit{ha})) = a\). Assume \(\forall a\, :\, \beta,\; \forall \mathit{ha}\, :\, a \in t,\; i(j(a,\allowbreak{} \mathit{ha}),\allowbreak{} \mathit{hj}(a,\allowbreak{} \mathit{ha})) = a\). \section{Degree sum}
\begin{lemma}[{Dart fst fiber}]\label{theorem:1}
Let \(V\) be a type. Let \(G\) be a simple graph on \(V\). Fix a finite enumeration of \(V\). Assume adjacency in \(G\) is decidable. Assume equality on \(V\) is decidable. Let \(v\) be an element of \(V\). Then
\begin{center}
\(\displaystyle \{d\in \operatorname{oriented\,edges}(G) \mid \operatorname{tail}(d) = v\} = (w\mapsto\operatorname{dart}_{G}(v,\allowbreak{} w))[N_{G}(v)].\)
\end{center}
\end{lemma}

\begin{proof}
For the required implication, argue as follows.
\begin{proofstep}
Fix \(d\) in \(\operatorname{oriented\,edges}(G)\).
The substitution uses the following argument.
\begin{proofstep}
Fix \(a\) in \(N_{G}(v)\).
By \paperlink{premise:46696e7365742e6d656d5f756e6976}{Prerequisite~P\ref{premise:46696e7365742e6d656d5f756e6976}} with \(\alpha = N_{G}(v)\) and \(x = a\), we obtain \(a \in N_{G}(v)\).
By \paperlink{premise:65715f74727565}{Prerequisite~P\ref{premise:65715f74727565}}, we obtain \(\left(a \in N_{G}(v)\right) = \mathrm{True}\).
Applying the same function to equal arguments, we obtain \((a \in N_{G}(v) \land \operatorname{dart}_{G}(v,\allowbreak{} a) = d) = (\mathrm{True} \land \operatorname{dart}_{G}(v,\allowbreak{} a) = d)\).
By \paperlink{premise:747275655f616e64}{Prerequisite~P\ref{premise:747275655f616e64}} with \(p = \operatorname{dart}_{G}(v, a) = d\), we obtain \((\mathrm{True} \land \operatorname{dart}_{G}(v,\allowbreak{} a) = d) = (\operatorname{dart}_{G}(v,\allowbreak{} a) = d)\).
Chaining the equalities, we obtain \((a \in N_{G}(v) \land \operatorname{dart}_{G}(v,\allowbreak{} a) = d) = (\operatorname{dart}_{G}(v,\allowbreak{} a) = d)\).
\end{proofstep}
Consider the following case.
\begin{proofstep}
Assume \(\operatorname{tail}(d) = v\).
Choose the witness \(\operatorname{head}(d)\).
The construction requires evidence that \(\operatorname{tail}(d) \sim_{G} \operatorname{head}(d)\).
By \paperlink{premise:53696d706c6547726170682e446172742e61646a}{Prerequisite~P\ref{premise:53696d706c6547726170682e446172742e61646a}} with \(\mathit{self} = d\), we obtain \(\operatorname{tail}(d) \sim_{G} \operatorname{head}(d)\).
By definition, we obtain \(\operatorname{tail}(\operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (\operatorname{head}(d)\, :\, N_{G}(\operatorname{tail}(d))))) = \operatorname{tail}(\operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (\operatorname{head}(d)\, :\, N_{G}(\operatorname{tail}(d)))))\).
By definition, we obtain \(\operatorname{head}(\operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (\operatorname{head}(d)\, :\, N_{G}(\operatorname{tail}(d))))) = \operatorname{head}(\operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (\operatorname{head}(d)\, :\, N_{G}(\operatorname{tail}(d)))))\).
By \paperlink{premise:50726f642e657874}{Prerequisite~P\ref{premise:50726f642e657874}}, we obtain \(\operatorname{to\,Prod}(\operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (\operatorname{head}(d)\, :\, N_{G}(\operatorname{tail}(d))))) = \operatorname{to\,Prod}(d)\).
By \paperlink{premise:53696d706c6547726170682e446172742e657874}{Prerequisite~P\ref{premise:53696d706c6547726170682e446172742e657874}} with \(d_{1} = \operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (\operatorname{head}(d)\, :\, N_{G}(\operatorname{tail}(d))))\) and \(d_{2} = d\), we obtain \(\operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (\operatorname{head}(d)\, :\, N_{G}(\operatorname{tail}(d)))) = d\).
This witness proves existence. Hence \(\exists h\text{ proving }(\operatorname{head}(d) \in N_{G}(\operatorname{tail}(d))),\; \operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (\operatorname{head}(d)\, :\, N_{G}(\operatorname{tail}(d)))) = d\).
This witness proves existence. Hence \(\exists x\, :\, V,\; \exists h\text{ proving }(x \in N_{G}(\operatorname{tail}(d))),\; \operatorname{dart}_{G}(\operatorname{tail}(d),\allowbreak{} (x\, :\, N_{G}(\operatorname{tail}(d)))) = d\).
Use the hypothesis \(\operatorname{tail}(d) = v\).
Substituting equals, we obtain \(\exists x\, :\, V,\; \exists h\text{ proving }(x \in N_{G}(v)),\; \operatorname{dart}_{G}(v,\allowbreak{} (x\, :\, N_{G}(v))) = d\).
\end{proofstep}
Consider the following case.
\begin{proofstep}
Assume \(\exists x\, :\, V,\; \exists h\text{ proving }(x \in N_{G}(v)),\; \operatorname{dart}_{G}(v,\allowbreak{} (x\, :\, N_{G}(v))) = d\).

\begin{theorem}[{Exists differs from odd degree}]\label{theorem:12}
Let \(V\) be a type. Let \(G\) be a simple graph on \(V\). Fix a finite enumeration of \(V\). Assume adjacency in \(G\) is decidable. Let \(v\) be an element of \(V\). Assume \(\operatorname{Odd}\left(\deg_{G}(v)\right)\). Then \(\exists w\, :\, V,\; w \ne v \land \operatorname{Odd}\left(\deg_{G}(w)\right)\).
\end{theorem}

\begin{proof}
By \paperlink{theorem:11}{Lemma~\ref{theorem:11}}, we obtain \(\operatorname{Odd}(|\{w\in V \mid w \ne v \land \operatorname{Odd}(\deg_{G}(w))\}|)\).

\begin{proofstep}
